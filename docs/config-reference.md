# gorca configuration reference

The file is `$XDG_CONFIG_HOME/gorca/gorca.toml` (default `~/.config/gorca/gorca.toml`), or the
path given by `--config=PATH` or `GORCA_CONFIG`. It is reloaded when it changes, except
`[speech] backend`, which needs a restart.

The listing below is generated by `gorca config print-default` from the settings table, so it
lists every key. `gorca config check` reports problems in your file.

## Speech in another language

`[speech] language` is the language code sent to the speech server. Empty means the language
of your locale (`LC_ALL`, `LC_MESSAGES`, then `LANG`; `ru_RU.UTF-8` gives `ru`), and `en` when
there is none. `output_module` and `voice` are sent only when set.

Russian with speech-dispatcher:

```toml
[speech]
backend = "ssip"
language = "ru"
output_module = "espeak-ng"   # or "rhvoice"
voice = ""                    # optional, a voice name your module offers
```

Russian with voicego (it accepts the same `SET self LANGUAGE`, `OUTPUT_MODULE` and
`SYNTHESIS_VOICE` commands):

```toml
[speech]
backend = "voicego"
language = "ru"
voice = ""                    # optional, a voice name from your voicego engine
```

## Default configuration

```toml
# gorca settings: every key at its default. Delete what you do not change.
# Keymap overrides go in [keymap.<context>] tables (see docs/config-reference.md).
schema = 1

[keyboard]
# Key bindings preset. (one of "nvda" or "orca")
preset = "nvda"
# Keyboard layout of the preset (keypad review keys on the desktop layout). (one of "desktop" or "laptop")
layout = "desktop"
# Key names that act as the gorca modifier. (list of strings)
modifiers = ["Insert", "KP_Insert", "Caps_Lock"]
# Drop Caps_Lock as a modifier when it switches keyboard layouts. (true or false)
caps_lock_layout_guard = true
# Window in milliseconds for double and triple key presses. (integer 100 to 2000)
multi_press_ms = 500
# Ignore a repeat of the same key within this many milliseconds. (integer 0 to 100)
double_tap_guard_ms = 15
# Which key presses cancel speech. (one of "non-modifier" or "never")
interrupt = "non-modifier"
# Control stops speech. (true or false)
stop_speech_on_control = true
# Shift pauses and resumes speech. (true or false)
pause_speech_on_shift = true

[keymap]
# Use only the [keymap.*] tables instead of layering them over the preset. (true or false)
replace_preset = false

[speech]
# Speech rate; 50 is the server default. (integer 0 to 100)
rate = 50
# Speech pitch; 50 is the server default. (integer 0 to 100)
pitch = 50
# Speech volume; 100 is the server volume. (integer 0 to 100)
volume = 100
# How much one rate, pitch or volume command changes the level. (integer 1 to 25)
rate_step = 5
# Synthesis voice name; empty is the server default (SSIP SYNTHESIS_VOICE). (string)
voice = ""
# Speech language code such as "ru"; empty is derived from LC_ALL/LC_MESSAGES/LANG, else "en". (string)
language = ""
# Speech-dispatcher output module such as "espeak-ng" or "rhvoice"; empty is the server default. (string)
output_module = ""
# Speech server: voicego with speech-dispatcher fallback (auto), voicego only, or speech-dispatcher only (ssip); the --speech flag wins; needs a restart. (one of "auto", "voicego" or "ssip")
backend = "auto"
# How much punctuation is spoken. (one of "none", "some", "most" or "all")
punctuation = "some"
# How a capital letter spoken alone is announced: any of "pitch", "sound" (earcon) and "word" ("cap" before the letter), for example ["pitch", "sound"]; [] is none. (list of strings)
caps = ["pitch"]
# How far the pitch of a capital moves from the voice, in percent of the pitch range (needs "pitch" in caps). (integer 0 to 100)
caps_pitch_change = 30

[echo]
# Speak each typed character. (true or false)
typed_characters = true
# Speak each typed word. (true or false)
typed_words = false
# Speak command keys such as Ctrl+S. (true or false)
command_keys = false
# Speak text inserted by paste or autocomplete. (true or false)
inserted_text = true

[text]
# Say all moves the caret as it reads. (true or false)
say_all_moves_caret = true
# Caret events within this many milliseconds of an edit count as part of it. (integer 20 to 1000)
caret_edge_ms = 100
# Caret events later than this many milliseconds after a key are not announced as a move. (integer 100 to 2000)
caret_late_ms = 400

[browse]
# Read a web page automatically when it finishes loading. (true or false)
say_all_on_load = true
# Say the description of a web control after a short pause. (true or false)
description_pause = true

[terminal]
# Wait this many milliseconds for terminal output to settle before speaking. (integer 10 to 200)
settle_ms = 30
# Output longer than this many lines is summarised instead of read. (integer 4 to 400)
burst_lines = 40
# Line endings that mark a shell prompt. (list of strings)
prompt_markers = ["$ ", "# ", "% ", "> ", "❯ ", "» "]
# Regular expression that matches a prompt; empty uses prompt_markers only. (string)
prompt_regex = ""
# Regular expression for lines never to speak, such as password prompts; empty is none. (string)
secret_regex = ""

[verbosity]
# How much detail is spoken. (one of "brief", "normal" or "verbose")
level = "normal"
# Full or brief confirmations of settings changes. (one of "full" or "brief")
confirmations = "full"
```
